# Cell 4 — Fonctions de l'étape 1
def read3d(path, pixel=sitk.sitkFloat32):
    img = sitk.ReadImage(str(path), pixel)
    if img.GetDimension() == 4:
        img = img[:, :, :, 0]
    return img

def resample_iso(img, spacing=(1.0, 1.0, 1.0), interp=sitk.sitkLinear):
    sp, sz = np.array(img.GetSpacing()), np.array(img.GetSize())
    new_sz = [int(round(s * p / n)) for s, p, n in zip(sz, sp, spacing)]
    return sitk.Resample(img, new_sz, sitk.Transform(), interp, img.GetOrigin(),
                         spacing, img.GetDirection(), 0.0, img.GetPixelID())

def _stage1(args):
    sid, flair_p, t1_p, roi_p, out_root, n_threads = args
    out = Path(out_root) / sid
    needed = [out / 'flair.nii.gz', out / 't1_reg.nii.gz', out / 'reg.json']
    if all(p.exists() for p in needed):
        return sid, True, 'cache'
    try:
        sitk.ProcessObject.SetGlobalDefaultNumberOfThreads(n_threads)
        out.mkdir(parents=True, exist_ok=True)
        info = {}
        # FLAIR : orientation commune + 1 mm isotrope
        fl = sitk.DICOMOrient(read3d(flair_p), 'LPS')
        fl = resample_iso(fl, SPACING, sitk.sitkLinear)
        # ROI -> grille FLAIR (plus proche voisin), contrôle du volume
        if isinstance(roi_p, str):
            roi = read3d(roi_p, sitk.sitkFloat32)
            roi_vol_orig = float((sitk.GetArrayViewFromImage(roi) > 0).sum() * np.prod(roi.GetSpacing()))
            roi_r = sitk.Resample(roi, fl, sitk.Transform(), sitk.sitkNearestNeighbor, 0.0, sitk.sitkFloat32)
            roi_r = sitk.Cast(roi_r > 0, sitk.sitkUInt8)
            roi_vol_new = float(sitk.GetArrayViewFromImage(roi_r).sum() * np.prod(SPACING))
            info['roi_ml'] = roi_vol_new / 1000
            info['roi_volume_ratio'] = roi_vol_new / max(roi_vol_orig, 1e-6)
            sitk.WriteImage(roi_r, str(out / 'roi.nii.gz'))
        # Recalage rigide T1 -> FLAIR
        t1 = read3d(t1_p)
        init = sitk.CenteredTransformInitializer(fl, t1, sitk.Euler3DTransform(),
                                                 sitk.CenteredTransformInitializerFilter.GEOMETRY)
        R = sitk.ImageRegistrationMethod()
        R.SetMetricAsMattesMutualInformation(numberOfHistogramBins=50)
        R.SetMetricSamplingStrategy(R.RANDOM)
        R.SetMetricSamplingPercentage(0.05, 42)
        R.SetInterpolator(sitk.sitkLinear)
        R.SetOptimizerAsRegularStepGradientDescent(learningRate=2.0, minStep=1e-4,
                                                   numberOfIterations=200, relaxationFactor=0.5)
        R.SetOptimizerScalesFromPhysicalShift()
        R.SetShrinkFactorsPerLevel([4, 2, 1])
        R.SetSmoothingSigmasPerLevel([2, 1, 0])
        R.SmoothingSigmasAreSpecifiedInPhysicalUnitsOn()
        R.SetInitialTransform(sitk.Euler3DTransform(), inPlace=False)
        info['mi_avant'] = float(R.MetricEvaluate(fl, t1))      # alignement par en-tête seul
        R.SetInitialTransform(init, inPlace=False)
        tx = R.Execute(fl, t1)
        info['mi_apres'] = float(R.GetMetricValue())
        c = fl.TransformContinuousIndexToPhysicalPoint([s / 2 for s in fl.GetSize()])
        info['deplacement_centre_mm'] = float(np.linalg.norm(np.array(tx.TransformPoint(c)) - np.array(c)))
        t1r = sitk.Resample(t1, fl, tx, sitk.sitkLinear, 0.0, sitk.sitkFloat32)
        sitk.WriteImage(fl, str(out / 'flair.nii.gz'))
        sitk.WriteImage(t1r, str(out / 't1_reg.nii.gz'))
        with open(out / 'reg.json', 'w') as f:
            json.dump(info, f)
        return sid, True, None
    except Exception as e:
        return sid, False, f'{type(e).__name__}: {e}'

def run_hdbet(inp, outp, tries=6, wait=60):
    # Au premier appel, HD-BET télécharge ses poids depuis Zenodo, parfois indisponible (erreur 504) :
    # on réessaie avec une attente croissante (1, 2, 4, 8, 16 min).
    for k in range(1, tries + 1):
        r = subprocess.run([HDBET, '-i', str(inp), '-o', str(outp)] + BET_OPTS, capture_output=True, text=True)
        produced = list(Path(outp).glob('*.nii.gz')) if Path(outp).is_dir() else []
        if r.returncode == 0 and produced:
            return BET_OPTS
        err = (r.stderr or r.stdout)[-400:]
        print(f'  tentative {k}/{tries} échouée :', err.strip().splitlines()[-1] if err.strip() else '?')
        if k < tries:
            print(f'  nouvelle tentative dans {wait//60} min...'); time.sleep(wait); wait *= 2
    raise RuntimeError('HD-BET a échoué après plusieurs tentatives (voir le message ci-dessus).')

# Cell 8 — Fonctions de l'étape 3 (N4 renforcé + indicateurs QC)
import hashlib

def n4_correct(img, mask, shrink=2):
    arr = np.maximum(sitk.GetArrayFromImage(img), 0) + 1.0        # N4 exige des valeurs positives
    pos = sitk.GetImageFromArray(arr.astype(np.float32)); pos.CopyInformation(img)
    f = sitk.N4BiasFieldCorrectionImageFilter()
    f.SetMaximumNumberOfIterations([100, 100, 100, 50])           # 4 niveaux
    f.SetConvergenceThreshold(1e-6)
    f.SetNumberOfControlPoints([6, 6, 6])                         # champ un peu plus souple
    f.Execute(sitk.Shrink(pos, [shrink] * 3), sitk.Shrink(mask, [shrink] * 3))
    log_bias = f.GetLogBiasFieldAsImage(pos)
    return sitk.GetArrayFromImage(pos / sitk.Exp(log_bias)).astype(np.float32)

def znorm(arr, m, lo=0.5, hi=99.5):
    a, b = np.percentile(arr[m], [lo, hi])
    x = np.clip(arr, a, b)
    x = (x - x[m].mean()) / (x[m].std() + 1e-6)
    x[~m] = 0
    return x.astype(np.float32)

def smooth_in_mask(x, m, sigma=10):
    w = ndimage.gaussian_filter(m.astype(np.float32), sigma)
    return ndimage.gaussian_filter(x * m, sigma) / np.maximum(w, 1e-3)

def inhomogeneity(x, m):
    # écart-type de l'image très lissée dans le cerveau (érodé) : élevé = dégradé d'intensité résiduel
    me = ndimage.binary_erosion(m, iterations=5)
    return float(smooth_in_mask(x, m)[me].std()) if me.any() else np.nan

def mi_in_mask(a, b, m):
    # information mutuelle (Mattes) FLAIR/T1 dans le cerveau, sur images corrigées : contrôle d'alignement
    A, B = sitk.GetImageFromArray(a.astype(np.float32)), sitk.GetImageFromArray(b.astype(np.float32))
    M = sitk.GetImageFromArray(m.astype(np.uint8))
    R = sitk.ImageRegistrationMethod()
    R.SetMetricAsMattesMutualInformation(numberOfHistogramBins=50)
    R.SetMetricFixedMask(M)
    R.SetMetricSamplingStrategy(R.RANDOM)
    R.SetMetricSamplingPercentage(0.2, 42)
    R.SetInterpolator(sitk.sitkLinear)
    R.SetInitialTransform(sitk.TranslationTransform(3), inPlace=False)
    return float(R.MetricEvaluate(A, B))

def md5(path):
    h = hashlib.md5()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()

def _stage3(args):
    sid, s1_root, final_root, n_threads = args
    s1, dst = Path(s1_root) / sid, Path(final_root) / f'{sid}.npz'
    qfile = Path(final_root) / f'{sid}_qc.json'
    if dst.exists() and qfile.exists():
        return sid, True, 'cache', json.load(open(qfile))
    try:
        sitk.ProcessObject.SetGlobalDefaultNumberOfThreads(n_threads)
        fl = sitk.ReadImage(str(s1 / 'flair.nii.gz'), sitk.sitkFloat32)
        t1 = sitk.ReadImage(str(s1 / 't1_reg.nii.gz'), sitk.sitkFloat32)
        mk = sitk.ReadImage(str(s1 / 'brainmask.nii.gz'), sitk.sitkUInt8)
        mk.CopyInformation(fl)
        m = sitk.GetArrayFromImage(mk) > 0
        flz, t1z = znorm(n4_correct(fl, mk), m), znorm(n4_correct(t1, mk), m)
        roi = (sitk.GetArrayFromImage(sitk.ReadImage(str(s1 / 'roi.nii.gz'))) > 0) if (s1 / 'roi.nii.gz').exists() \
              else np.zeros_like(m)
        idx = np.argwhere(m)
        lo = np.maximum(idx.min(0) - 4, 0); hi = np.minimum(idx.max(0) + 5, m.shape)
        sl = tuple(slice(int(a), int(b)) for a, b in zip(lo, hi))
        mc = m[sl]
        q = {'brain_ml': float(m.sum() * np.prod(SPACING) / 1000),
             'roi_dans_cerveau': float(np.logical_and(roi, m).sum() / roi.sum()) if roi.sum() else None,
             'mi_final': mi_in_mask(flz[sl], t1z[sl], mc),
             'inhom_t1': inhomogeneity(t1z[sl], mc),
             'inhom_flair': inhomogeneity(flz[sl], mc),
             'forme_recadree': 'x'.join(map(str, mc.shape))}
        np.savez_compressed(dst, flair=flz[sl].astype(np.float16), t1=t1z[sl].astype(np.float16),
                            mask=mc.astype(np.uint8), roi=roi[sl].astype(np.uint8),
                            bbox=np.stack([lo, hi]), spacing=np.array(SPACING))
        q['md5'] = md5(dst)
        with open(qfile, 'w') as f:
            json.dump(q, f)
        return sid, True, None, q
    except Exception as e:
        return sid, False, f'{type(e).__name__}: {e}', None
